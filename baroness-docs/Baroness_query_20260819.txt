/* ==================================================
[공통] 1. 포탈 사용자 ID 조회
==================================================
- 의도: 로그인한 유저 정보로 시스템 내부 고유 ID(portal_user_id)를 식별할 때 사용
- 입력 변수: ${USER_ID} (현재 웹사이트 로그인 세션 유저 ID)
- 출력 결과: violet.fn_get_group_hierarchy_with_users_new() 등에 입력할 내부 ID
*/

SELECT
	user_id
FROM
	violet.tb_users_sso_accounts
WHERE
	company = 'baroness'
	AND provider_user_id = '${USER_ID}'
;


/* ==================================================
[차량 데이터] 2. 실시간 차량 상태 조회
==================================================
- 입력 변수: ${PORTAL_USER_ID.results[0][0]} ('포탈 사용자 ID 조회' 쿼리 결과로 나온 포탈 유저 ID)
- 주요 로직: 
  1. 유저 권한 계층(Hierarchy)에 속한 차량 및 View Only 차량 식별
  2. 가장 최근 수집된 데이터 시간과 현재 시간을 비교하여 10초 이내 데이터면 'online', 지났으면 'offline' 처리
  3. 시차(Timezone)를 고려하여 현지 수집 시간 계산
- 출력 컬럼 안내:
  - op_mode: 차량 작동 모드 (오프라인 시 '-')
  - condition / check: 실시간 온/오프라인 상태 (online / offline)
  - notes / note: 보기 권한만 있는 차량은 'View Only' 표시
*/

WITH device_user AS (
    SELECT sensor_id, user_id, 'user' AS src
    FROM violet.v_tb_device_user
    WHERE user_id IN (
        SELECT user_id
        FROM violet.fn_get_group_hierarchy_with_users_new('${PORTAL_USER_ID.results[0][0]}')
    )
    UNION ALL
    SELECT sensor_id, user_id, 'exc' AS src
    FROM violet.tb_device_user_exc
    WHERE user_id = '${PORTAL_USER_ID.results[0][0]}'
),
collect_info AS (
    SELECT
        ci.vehicle_id,
        MAX(ci.last_collect_datetime) FILTER (WHERE ci.topic = 'condition') AS last_condition_dt,
        MAX(ci.last_collect_datetime) FILTER (WHERE ci.topic = 'log') AS last_log_dt
    FROM public.bms_collect_info ci
      JOIN device_user du
      ON du.sensor_id = ci.vehicle_id
    GROUP BY ci.vehicle_id
),
latest_condition AS (
    SELECT
        c.collect_datetime,
        COALESCE(c.timezone, 9) as timezone,
        c.vehicle_id,
        c.op_mode_m AS op_mode
    FROM bms_condition c
    JOIN collect_info ci
      ON ci.vehicle_id = c.vehicle_id
      AND ci.last_condition_dt = c.collect_datetime
),
latest_log AS (
    SELECT
        l.vehicle_id,
        l.eg_time,
        l.uman_time
    FROM bms_log l
    JOIN collect_info ci
      ON ci.vehicle_id = l.vehicle_id
      AND ci.last_log_dt = l.collect_datetime
),
user_groups AS (
    SELECT
        du.sensor_id AS sensor_id,
        gm.group_nm AS group_nm,
        du.user_id
    FROM violet.tb_device_user du
    LEFT JOIN violet.tb_users u
      ON du.user_id = u.user_id
    LEFT JOIN violet.tb_group_mgmt gm
      ON u.group_id::int = gm.id
    WHERE du.user_id IN (
        SELECT user_id
        FROM violet.tb_users
        WHERE role_id = 'ROLE_GROUP'
    )
)
SELECT DISTINCT
    (b.collect_datetime + (COALESCE(b.timezone, 9) || ' hours')::INTERVAL) AS collect_datetime,
    CASE
        WHEN (b.collect_datetime AT TIME ZONE 'UTC') < now() - INTERVAL '10 seconds' THEN '-'
        ELSE b.op_mode
        END AS op_mode,
    b.vehicle_id AS sensor_id,
    v.specification,
    ug.group_nm,
    l.eg_time,
    l.uman_time,
    CASE
        WHEN (b.collect_datetime AT TIME ZONE 'UTC') >= now() - INTERVAL '10 seconds' THEN 'online'
        ELSE 'offline'
        END AS condition,
    CASE
        WHEN (b.collect_datetime AT TIME ZONE 'UTC') >= now() - INTERVAL '10 seconds' THEN 'online'
        ELSE 'offline'
    END AS check,
    (b.collect_datetime + (COALESCE(b.timezone, 9) || ' hours')::INTERVAL) AS data_time,
    b.vehicle_id,
    CASE WHEN BOOL_OR(du.src = 'exc') THEN 'View Only' ELSE '' END AS notes,
    CASE WHEN BOOL_OR(du.src = 'exc') THEN 'View Only' ELSE '' END AS note
FROM latest_condition b
LEFT JOIN vehicle_info v
  ON b.vehicle_id = v.vehicle_id
LEFT JOIN latest_log l
  ON b.vehicle_id = l.vehicle_id
LEFT JOIN user_groups ug
  ON b.vehicle_id = ug.sensor_id
LEFT JOIN device_user du
  ON b.vehicle_id = du.sensor_id
GROUP BY
    b.collect_datetime, COALESCE(b.timezone, 9), b.op_mode, b.vehicle_id,
    v.specification,
    ug.group_nm,
    l.eg_time, l.uman_time
    ;
    
    
 /* ==================================================
[차량 상세] 3. 특정 차량 정보 및 위치/통계 상세 조회
==================================================
- 입력 변수: 
  - ${vehicle_id}: 선택한 차량 고유 ID
  - ${data_time}: 기준 조회 일시
  - ${date_time_range_picker_1.startDate / endDate}: 조회 시작/종료 시간 범위
*/

-- --------------------------------------------------
-- [공통] 3-1. 차량의 timezone 조회 (vehicle_timezone)
-- 의도: 선택한 차량의 최신 수집 데이터 기준 Timezone 정보를 조회하여 하위 쿼리에서 사용
-- 입력 변수: ${vehicle_id}
-- --------------------------------------------------
SELECT 
    COALESCE(timezone, 9) as tz
FROM bms_log
WHERE vehicle_id = '${vehicle_id}'
ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [지도] 3-2. 차량의 마지막 수집 위치 조회
-- 로직: 기준 시간(${data_time}) 전후 1분 이내의 최신 위경도 데이터 1건 추출
-- --------------------------------------------------
SELECT 
    collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval AS "Data Time",
    lat, lon
FROM bms_condition
WHERE
    vehicle_id = '${vehicle_id}'
    AND lat IS NOT NULL AND lon IS NOT NULL
    AND lat <> 0 AND lon <> 0
    AND collect_datetime
        BETWEEN '${data_time}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval - INTERVAL '1 minutes'
        AND     '${data_time}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval + INTERVAL '1 minutes'
ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [지도] 3-3. 특정 기간 차량 이동 경로 조회
-- --------------------------------------------------
SELECT
    collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval AS "Data Time",
    lat, lon
FROM bms_condition
WHERE
    vehicle_id = '${vehicle_id}'
    AND lat IS NOT NULL AND lon IS NOT NULL
    AND lat <> 0 AND lon <> 0
    AND collect_datetime >= '${date_time_range_picker_1.startDate}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval
    AND collect_datetime <= '${date_time_range_picker_1.endDate}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval
ORDER BY collect_datetime ASC
;

-- --------------------------------------------------
-- [지도] 3-4. 차량의 이전 위치 조회
-- --------------------------------------------------
SELECT 
    collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval AS "Data Time",
    lat, lon
FROM bms_condition
WHERE
    vehicle_id = '${vehicle_id}'
    AND lat IS NOT NULL AND lon IS NOT NULL
    AND lat <> 0 AND lon <> 0
    AND collect_datetime
        BETWEEN '${before}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval - INTERVAL '1 minutes'
        AND     '${before}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval + INTERVAL '1 minutes'
ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [테이블] 3-5. 차량 상태 상세 정보 테이블 조회
-- 출력 주요 데이터: 운전/시스템모드, 연료잔량(fuel_lv), 속도(sog), 재생진행도(pb_prg), 경고내용(alert_m) 등
-- --------------------------------------------------
SELECT 
	collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval as collect_datetime, 
	op_mode_m as op_mode, 
	sys_mode_m as sys_mode, 
	uni_posi_m as uni_posi,  
	fuel_lv, 
	sog, 
	pb_data, 
	pb_prg, 
	pb_est, 
	alert_m, 
	lat, 
	lon
FROM bms_condition
WHERE 
	vehicle_id = '${vehicle_id}'
	--AND lat IS NOT NULL AND lon IS NOT NULL
    --AND lat <> 0 AND lon <> 0	
	AND collect_datetime 
		BETWEEN ('${data_time}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval) - INTERVAL '1 minutes'
        AND ('${data_time}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval) + INTERVAL '1 minutes'
ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [차트] 3-6. 분 단위 차량 핵심 데이터 시계열 차트 생성
-- 로직: 설정된 기간 동안 데이터가 없는 분(Minute) 단위 영역도 0값으로 채우도록 generate_series 선행 결합
-- --------------------------------------------------
SELECT 
	DataTime, 
	sum(fuel_lv) as fuel_lv, 
	sum(sog) as sog, 
	sum(pb_prg) as pb_prg, 
	sum(pb_est) as pb_est
FROM (
	SELECT 
		generate_series(
			TIMESTAMP '${date_time_range_picker_1.startDate}',
			TIMESTAMP '${date_time_range_picker_1.endDate}', 
			INTERVAL '1 minute'
		) as DataTime, 
		0 as fuel_lv, 0 as sog, 0 as pb_prg, 0 as pb_est
	UNION ALL 
	SELECT 
		(collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval) as DataTime, 
		max(fuel_lv) as fuel_lv, 
		max(sog) as sog, 
		max(pb_prg) as pb_prg, 
		max(pb_est) as pb_est
	FROM bms_condition
	WHERE 
		collect_datetime >= ('${date_time_range_picker_1.startDate}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval)
		AND collect_datetime <= ('${date_time_range_picker_1.endDate}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval)
		AND vehicle_id = '${vehicle_id}'
	GROUP BY collect_datetime
	) a
GROUP BY DataTime
ORDER BY DataTime;
;

- 주요 로직:
  1. 유저 접근 가능 차량 및 예외 차량(View Only) 통합 분류 (device_user_summary)
  2. 선택한 기간 조건 내의 유효 경고 메시지만 초 단위 정밀도로 중복 제거하여 집계 (latest_alert)
  3. locale 조건에 따라 결과 데이터 컬럼명을 다국어로 자동 변환하여 출력

WITH device_user_summary AS (
    SELECT 
        sensor_id,
        BOOL_OR(src = 'exc') AS is_exc
    FROM (
        SELECT sensor_id, 'user' AS src 
        FROM violet.v_tb_device_user 
        WHERE user_id IN (
            SELECT user_id FROM violet.fn_get_group_hierarchy_with_users_new('${PORTAL_USER_ID.results[0][0]}')
        )
        UNION ALL
        SELECT sensor_id, 'exc' AS src 
        FROM violet.tb_device_user_exc 
        WHERE user_id = '${PORTAL_USER_ID.results[0][0]}'
    ) combined
    GROUP BY sensor_id
),
user_group AS (
    SELECT DISTINCT ON (du.sensor_id)
        du.sensor_id,
        gm.group_nm
    FROM violet.tb_device_user du
    JOIN violet.tb_users u 
    	ON du.user_id = u.user_id 
    	AND u.role_id = 'ROLE_GROUP'
    LEFT JOIN violet.tb_group_mgmt gm 
    	ON u.group_id::int = gm.id
    ORDER BY du.sensor_id, gm.group_nm
),
latest_alert AS (
    SELECT DISTINCT
        a.vehicle_id,
        a.alert_m,
        a.op_mode_m,
        date_trunc(
            'second',
            a.collect_datetime + (COALESCE(a.timezone, 9) || ' hours')::INTERVAL
        ) AS collect_datetime,
        dus.is_exc
    FROM bms_alert a
    JOIN device_user_summary dus 
    	ON a.vehicle_id = dus.sensor_id
    WHERE 
        a.collect_datetime >= ('${date_picker_1}'::timestamp - (COALESCE(a.timezone, 9)::text || ' hours')::interval)
        AND a.collect_datetime <= ('${date_picker_2}'::timestamp - (COALESCE(a.timezone, 9)::text || ' hours')::interval)
        AND a.alert_m <> '' AND a.alert_m <> 'null'
)
SELECT
    b.collect_datetime as "{% if locale == 'ja' %}データ時間{% elif locale == 'de' %}Datenzeit{% elif locale == 'ko' %}데이터 시간{% else %}Data Time{% endif %}",
    b.op_mode_m AS "{% if locale == 'ja' %}動作モード{% elif locale == 'de' %}Betriebsmodus{% elif locale == 'ko' %}작동 모드{% else %}Operation Mode{% endif %}",
    b.vehicle_id as "{% if locale == 'ja' %}車両 ID{% elif locale == 'de' %}Fahrzeug ID{% elif locale == 'ko' %}차량 ID{% else %}Vehicle ID{% endif %}",
    ug.group_nm as "{% if locale == 'ja' %}所有者{% elif locale == 'de' %}Eigentümer{% elif locale == 'ko' %}소유자{% else %}Owner{% endif %}",
    b.alert_m as "{% if locale == 'ja' %}警告メッセージ{% elif locale == 'de' %}Warnmeldung{% elif locale == 'ko' %}경고 메시지{% else %}Alert Message{% endif %}",
    b.collect_datetime as data_time,
    b.vehicle_id as vehicle_id,
    ug.group_nm as group_nm,
    b.alert_m as alert_m,
    CASE WHEN b.is_exc THEN 'View Only' ELSE '' END AS "{% if locale == 'ja' %}備考{% elif locale == 'de' %}Bemerkungen{% elif locale == 'ko' %}비고{% else %}Notes{% endif %}",
    CASE WHEN b.is_exc THEN 'View Only' ELSE '' END AS note
FROM latest_alert b
LEFT JOIN user_group ug 
	ON b.vehicle_id = ug.sensor_id
ORDER BY b.collect_datetime DESC
;
  
  
/* ==================================================
[경고 상세] 5. 선택한 경고 알림의 세부 정보 조회
==================================================
- 입력 변수:
  - ${vehicle_id}: 선택한 차량 고유 ID
  - ${slt_event_dt}: 유저가 클릭하거나 선택한 경고 발생 시점 (이벤트 일시)
  - ${alert} 또는 ${alert_m}: 확인할 특정 경고 메시지 내용
*/

-- --------------------------------------------------
-- [테이블] 5-1. 경고 발생 순간의 차량 기본 상태 정보
-- 로직: 경고 발생 시점(${slt_event_dt}) 전후 1초 이내의 가장 최신 bms_alert 데이터 1건 추출
-- --------------------------------------------------
SELECT 
	(collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval) as collect_datetime, 
	op_mode_m as op_mode, 
	sys_mode_m as sys_mode, 
	uni_posi_m as uni_posi,  
	fuel_lv, 
	sog, 
	pb_data, 
	pb_prg, 
	pb_est, 
	alert_m, 
	lat, 
	lon
FROM bms_alert
WHERE 
	collect_datetime
		BETWEEN '${slt_event_dt}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval - INTERVAL '1 seconds' 
		AND  '${slt_event_dt}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval + INTERVAL '1 seconds'
	AND vehicle_id = '${vehicle_id}'
	AND alert_m = '${alert}'
	AND alert_m <> '' and alert_m <> 'null'
ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [테이블] 5-2. 경고 발생 순간의 하드웨어/제어계 상세 정보
-- 출력 주요 데이터: 엔진속도(eg_rpm), 기울기(roll, pitch), 위성수(num_sv), 전압(bt_volt), 타각지시/피드백(ster_ins, ster_fb) 등
-- --------------------------------------------------
SELECT 
	(collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval) as collect_datetime,
	eg_time, 
	eg_rpm, 
	roll, 
	pitch, 
	oside_temp, 
	quality_m as quality, 
	head_type_m as head_type,  
	num_sv, 
	age, 
	hdop, 
	bt_volt, 
	eg_wtemp, 
	ster_ins, 
	ster_fb, 
	trv_ins, 
	trv_fb 
FROM bms_alert
WHERE 
	collect_datetime 
		BETWEEN '${slt_event_dt}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval - INTERVAL '1 seconds' 
		AND  '${slt_event_dt}'::timestamp - (${vehicle_timezone.results[0][0]} || ' hours')::interval + INTERVAL '1 seconds'
	AND vehicle_id = '${vehicle_id}'
	AND alert_m = '${alert}'
	AND alert_m <> '' and alert_m <> 'null'
	ORDER BY collect_datetime DESC
LIMIT 1
;

-- --------------------------------------------------
-- [테이블] 5-3. 이 차량에서 발생한 전체 경고 이력 타임라인 (중복 제거)
-- 로직: 
--   1. 초(Second) 단위 자른 시간 및 경고 메시지로 `ROW_NUMBER()` 중복 추출
--   2. 동일 초 내 동일 경고 수집 시 최신 1건만 남기고 다국어 컬럼 헤더 적용
-- --------------------------------------------------
SELECT
  date_trunc('second', collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval) 
    AS "{% if locale == 'ja' %}データ時間{% elif locale == 'de' %}Datenzeit{% elif locale == 'ko' %}데이터 시간{% else %}Data Time{% endif %}",
  alert_m 
    AS "{% if locale == 'ja' %}警告メッセージ{% elif locale == 'de' %}Warnmeldung{% elif locale == 'ko' %}경고 메시지{% else %}Alert Message{% endif %}",
  (collect_datetime + (${vehicle_timezone.results[0][0]} || ' hours')::interval) AS event_dt,
  lat,
  lon,
  alert_m
FROM (
  SELECT
    collect_datetime,
    alert_m,
    lat,
    lon,
    ROW_NUMBER() OVER (
      PARTITION BY date_trunc('second', collect_datetime), alert_m
      ORDER BY collect_datetime DESC
    ) AS rn
  FROM bms_alert
  WHERE vehicle_id = '${vehicle_id}'
    AND alert_m IS NOT NULL
    AND alert_m <> ''
    AND alert_m <> 'null'
) sub
WHERE rn = 1
ORDER BY collect_datetime DESC
;


/* ==================================================
[차량 관리] 6. 전체 차량 목록 및 소속 그룹 조회
==================================================
- 입력 변수: ${PORTAL_USER_ID.results[0][0]} (로그인 유저의 포탈 고유 ID)
- 로직: 
  1. 유저 하위 계층 그룹에 매핑된 차량 목록 추출
  2. 어떤 그룹에도 소속되지 않은(tb_device_user에 없는) 미할당 차량 목록을 추출하여 UNION 합산
*/

WITH user_groups AS (
    SELECT 
        du.sensor_id AS sensor_id, 
        gm.group_nm AS group_nm
    FROM violet.tb_device_user du
    LEFT JOIN violet.tb_users u 
        ON du.user_id = u.user_id
    LEFT JOIN violet.tb_group_mgmt gm 
        ON u.group_id::int = gm.id
    WHERE du.user_id IN (
        SELECT user_id FROM violet.tb_users WHERE role_id = 'ROLE_GROUP'
    )
)
SELECT
    v.vehicle_id as "차량 ID", -- 다국어 locale 대상 (ja: 車両 ID, de: Fahrzeug ID, en: Vehicle ID)
    v.specification as "사양", -- 다국어 locale 대상 (ja: 仕様, de: Spezifikation, en: Specification)
    v.model as "모델", -- 다국어 locale 대상 (ja: モデル, de: Modell, en: Model)
    v.function_limit as "기능 제한", -- 다국어 locale 대상 (ja: 機能制限, de: Funktionsbegrenzung, en: Function Limit)
    g.group_nm as "그룹명", -- 다국어 locale 대상 (ja: グループ名, de: Gruppenname, en: Group Name)
    v.vehicle_id 
FROM violet.tb_device_user du
LEFT JOIN vehicle_info v
    ON v.vehicle_id = du.sensor_id
LEFT JOIN user_groups g
    ON v.vehicle_id = g.sensor_id
WHERE du.user_id IN (
    SELECT user_id
    FROM violet.fn_get_group_hierarchy_with_users_new('${PORTAL_USER_ID.results[0][0]}')
)
	AND v.vehicle_id is not null
UNION 
SELECT
    v.vehicle_id as "차량 ID",
    v.specification as "사양",
    v.model as "모델",
    v.function_limit as "기능 제한",
    g.group_nm as "그룹명",
    v.vehicle_id
FROM vehicle_info v
LEFT JOIN user_groups g
    ON v.vehicle_id = g.sensor_id
WHERE v.vehicle_id NOT IN (
    SELECT distinct du.sensor_id
    FROM violet.tb_device_user du
)
	AND v.vehicle_id is not null
ORDER BY 1
;
